type CreateDraftInput struct {
	NamespaceCode NamespaceCode
	Actor         Actor
}

type createDraftFingerprintPayload struct {
	NamespaceCode NamespaceCode `json:"namespace_code"`
}

// CreateDraft creates the next draft version for a canonical taxonomy namespace.
//
// Version allocation is serialized by locking the namespace row. The command,
// taxonomy mutation, change evidence and outbox event all commit atomically.
func (s *Service) CreateDraft(
	ctx context.Context,
	input CreateDraftInput,
) (Version, error) {
	if s == nil ||
		s.repo == nil ||
		input.NamespaceCode == "" ||
		input.Actor.ActorRef == "" ||
		input.Actor.CommandRef == "" {
		return Version{}, ErrInvalidArgument
	}

	fingerprint, err := requestFingerprint(createDraftFingerprintPayload{
		NamespaceCode: input.NamespaceCode,
	})
	if err != nil {
		return Version{}, err
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return Version{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	namespace, err := s.repo.LockNamespaceByCode(
		ctx,
		tx,
		input.NamespaceCode,
	)
	if err != nil {
		return Version{}, err
	}

	if namespace.Status != NamespaceStatusActive {
		return Version{}, ErrConflict
	}

	replayed, err := s.replayVersionCommand(
		ctx,
		tx,
		namespace.ID,
		input.Actor.CommandRef,
		commandCreateDraft,
		fingerprint,
	)
	if err == nil {
		if err = tx.Commit(ctx); err != nil {
			return Version{}, err
		}
		return replayed, nil
	}
	if !errors.Is(err, ErrNotFound) {
		return Version{}, err
	}

	versionNumber, err := s.repo.NextVersionNumber(
		ctx,
		tx,
		namespace.ID,
	)
	if err != nil {
		return Version{}, err
	}

	version, err := s.repo.InsertVersion(
		ctx,
		tx,
		Version{
			ID:                "ctv_" + uuid.NewString(),
			NamespaceID:       namespace.ID,
			VersionNumber:     versionNumber,
			Status:            VersionStatusDraft,
			CreatedByActorRef: input.Actor.ActorRef,
		},
	)
	if err != nil {
		return Version{}, err
	}

	if err = s.repo.InsertChangeHistory(
		ctx,
		tx,
		ChangeHistoryEntry{
			ID:            "ctchg_" + uuid.NewString(),
			NamespaceID:   namespace.ID,
			VersionID:     version.ID,
			ActorRef:      input.Actor.ActorRef,
			CommandRef:    input.Actor.CommandRef,
			ChangeType:    changeDraftCreated,
			AggregateType: aggregateVersion,
			AggregateRef:  version.ID,
			BeforeState:   nil,
			AfterState:    version,
			CorrelationID: input.Actor.CorrelationID,
			CausationID:   input.Actor.CausationID,
		},
	); err != nil {
		return Version{}, err
	}

	if s.publisher != nil {
		if err = s.publisher.Publish(
			ctx,
			tx,
			eventDraftCreated,
			aggregateVersion,
			version.ID,
			version.Revision,
			"",
			input.Actor.CorrelationID,
			input.Actor.CausationID,
			map[string]any{
				"namespace_code": input.NamespaceCode,
				"namespace_id":   namespace.ID,
				"version_id":     version.ID,
				"version_number": version.VersionNumber,
				"status":         version.Status,
			},
		); err != nil {
			return Version{}, err
		}
	}

	if err = s.repo.InsertCommand(
		ctx,
		tx,
		CommandRecord{
			ID:                 "ctcmd_" + uuid.NewString(),
			NamespaceID:        namespace.ID,
			CommandRef:         input.Actor.CommandRef,
			CommandType:        commandCreateDraft,
			RequestFingerprint: fingerprint,
			ResultType:         resultTypeVersion,
			ResultRef:          version.ID,
			ResultRevision:     version.Revision,
		},
	); err != nil {
		return Version{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return Version{}, err
	}

	return version, nil
}

func (s *Service) replayVersionCommand(
	ctx context.Context,
	tx pgx.Tx,
	namespaceID string,
	commandRef string,
	commandType string,
	fingerprint string,
) (Version, error) {
	record, err := s.repo.GetCommandTx(
		ctx,
		tx,
		namespaceID,
		commandRef,
	)
	if err != nil {
		return Version{}, err
	}

	if record.CommandType != commandType ||
		record.RequestFingerprint != fingerprint ||
		record.ResultType != resultTypeVersion {
		return Version{}, ErrIdempotency
	}

	version, err := s.repo.GetVersionByIDTx(
		ctx,
		tx,
		record.ResultRef,
		false,
	)
	if err != nil {
		return Version{}, err
	}

	if version.NamespaceID != namespaceID {
		return Version{}, ErrConflict
	}

	return version, nil
}

func requestFingerprint(value any) (string, error) {
	raw, err := json.Marshal(value)
	if err != nil {
		return "", ErrInvalidArgument
	}

	sum := sha256.Sum256(raw)

	return hex.EncodeToString(sum[:]), nil
}

// CloneActiveVersion creates a new DRAFT from the currently published version.
//
// Facets and nodes receive new row identifiers while canonical codes,
// hierarchy, statuses and metadata are preserved. Parent links are remapped
// to the cloned node identifiers.
//
// The namespace row serializes version allocation and publication lifecycle.
func (s *Service) CloneActiveVersion(
	ctx context.Context,
	input CloneActiveVersionInput,
) (Version, error) {
	if s == nil ||
		s.repo == nil ||
		input.NamespaceCode == "" ||
		input.Actor.ActorRef == "" ||
		input.Actor.CommandRef == "" {
		return Version{}, ErrInvalidArgument
	}

	fingerprint, err := requestFingerprint(struct {
		NamespaceCode NamespaceCode `json:"namespace_code"`
	}{
		NamespaceCode: input.NamespaceCode,
	})
	if err != nil {
		return Version{}, err
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return Version{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	namespace, err := s.repo.LockNamespaceByCode(
		ctx,
		tx,
		input.NamespaceCode,
	)
	if err != nil {
		return Version{}, err
	}

	if namespace.Status != NamespaceStatusActive {
		return Version{}, ErrConflict
	}

	// Idempotent replay is checked before looking at the current publication.
	// The same command reference must always resolve to its original result.
	replayed, err := s.replayVersionCommand(
		ctx,
		tx,
		namespace.ID,
		input.Actor.CommandRef,
		commandCloneActiveVersion,
		fingerprint,
	)
	if err == nil {
		if err = tx.Commit(ctx); err != nil {
			return Version{}, err
		}
		return replayed, nil
	}
	if !errors.Is(err, ErrNotFound) {
		return Version{}, err
	}

	published, err := s.repo.GetPublishedVersionTx(
		ctx,
		tx,
		namespace.ID,
		false,
	)
	if err != nil {
		return Version{}, err
	}

	versionNumber, err := s.repo.NextVersionNumber(
		ctx,
		tx,
		namespace.ID,
	)
	if err != nil {
		return Version{}, err
	}

	sourceVersionID := published.ID

	draft, err := s.repo.InsertVersion(
		ctx,
		tx,
		Version{
			ID:                "ctv_" + uuid.NewString(),
			NamespaceID:       namespace.ID,
			VersionNumber:     versionNumber,
			Status:            VersionStatusDraft,
			SourceVersionID:   &sourceVersionID,
			CreatedByActorRef: input.Actor.ActorRef,
		},
	)
	if err != nil {
		return Version{}, err
	}

	facetCount, nodeCount, err := s.cloneVersionContent(
		ctx,
		tx,
		published,
		draft,
	)
	if err != nil {
		return Version{}, err
	}

	if err = s.repo.InsertChangeHistory(
		ctx,
		tx,
		ChangeHistoryEntry{
			ID:            "ctchg_" + uuid.NewString(),
			NamespaceID:   namespace.ID,
			VersionID:     draft.ID,
			ActorRef:      input.Actor.ActorRef,
			CommandRef:    input.Actor.CommandRef,
			ChangeType:    changeDraftCloned,
			AggregateType: aggregateVersion,
			AggregateRef:  draft.ID,
			BeforeState:   published,
			AfterState:    draft,
			CorrelationID: input.Actor.CorrelationID,
			CausationID:   input.Actor.CausationID,
		},
	); err != nil {
		return Version{}, err
	}

	if s.publisher != nil {
		if err = s.publisher.Publish(
			ctx,
			tx,
			eventDraftCloned,
			aggregateVersion,
			draft.ID,
			draft.Revision,
			"",
			input.Actor.CorrelationID,
			input.Actor.CausationID,
			map[string]any{
				"namespace_code":    input.NamespaceCode,
				"namespace_id":      namespace.ID,
				"source_version_id": published.ID,
				"source_version":    published.VersionNumber,
				"draft_version_id":  draft.ID,
				"draft_version":     draft.VersionNumber,
				"facet_count":       facetCount,
				"node_count":        nodeCount,
			},
		); err != nil {
			return Version{}, err
		}
	}

	if err = s.repo.InsertCommand(
		ctx,
		tx,
		CommandRecord{
			ID:                 "ctcmd_" + uuid.NewString(),
			NamespaceID:        namespace.ID,
			CommandRef:         input.Actor.CommandRef,
			CommandType:        commandCloneActiveVersion,
			RequestFingerprint: fingerprint,
			ResultType:         resultTypeVersion,
			ResultRef:          draft.ID,
			ResultRevision:     draft.Revision,
		},
	); err != nil {
		return Version{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return Version{}, err
	}

	return draft, nil

func (s *Service) cloneVersionContent(
	ctx context.Context,
	tx pgx.Tx,
	source Version,
	target Version,
) (int, int, error) {
	sourceFacets, err := s.repo.ListFacetsTx(
		ctx,
		tx,
		source.ID,
	)
	if err != nil {
		return 0, 0, err
	}

	facetIDMap := make(map[string]string, len(sourceFacets))

	for _, sourceFacet := range sourceFacets {
		newID := "ctf_" + uuid.NewString()
		facetIDMap[sourceFacet.ID] = newID

		_, err = s.repo.InsertFacet(
			ctx,
			tx,
			Facet{
				ID:            newID,
				NamespaceID:   target.NamespaceID,
				VersionID:     target.ID,
				Code:          sourceFacet.Code,
				DisplayName:   sourceFacet.DisplayName,
				Description:   sourceFacet.Description,
				StructureType: sourceFacet.StructureType,
				Position:      sourceFacet.Position,
				Status:        sourceFacet.Status,
				Metadata:      sourceFacet.Metadata,
			},
		)
		if err != nil {
			return 0, 0, err
		}
	}

	sourceNodes, err := s.repo.ListNodesTx(
		ctx,
		tx,
		source.ID,
	)
	if err != nil {
		return 0, 0, err
	}

	nodeIDMap := make(map[string]string, len(sourceNodes))

	// Allocate all cloned identifiers first so parent relationships can be
	// deterministically remapped before insertion.
	for _, sourceNode := range sourceNodes {
		nodeIDMap[sourceNode.ID] = "ctn_" + uuid.NewString()
	}

	inserted := make(map[string]bool, len(sourceNodes))
	remaining := len(sourceNodes)

	for remaining > 0 {
		progress := false

		for _, sourceNode := range sourceNodes {
			if inserted[sourceNode.ID] {
				continue
			}

			var clonedParentID *string

			if sourceNode.ParentID != nil {
				parentID, exists := nodeIDMap[*sourceNode.ParentID]
				if !exists {
					return 0, 0, ErrConflict
				}

				// Parent must already exist because the composite FK is
				// enforced immediately by PostgreSQL.
				if !inserted[*sourceNode.ParentID] {
					continue
				}

				clonedParentID = &parentID
			}

			clonedFacetID, exists := facetIDMap[sourceNode.FacetID]
			if !exists {
				return 0, 0, ErrConflict
			}

			_, err = s.repo.InsertNode(
				ctx,
				tx,
				Node{
					ID:            nodeIDMap[sourceNode.ID],
					NamespaceID:   target.NamespaceID,
					VersionID:     target.ID,
					FacetID:       clonedFacetID,
					CanonicalCode: sourceNode.CanonicalCode,
					DisplayName:   sourceNode.DisplayName,
					Description:   sourceNode.Description,
					ParentID:      clonedParentID,
					Position:      sourceNode.Position,
					Status:        sourceNode.Status,
					Metadata:      sourceNode.Metadata,
				},
			)
			if err != nil {
				return 0, 0, err
			}

			inserted[sourceNode.ID] = true
			remaining--
			progress = true
		}

		if !progress {
			// A healthy published taxonomy cannot reach this point. It means
			// the source hierarchy contains an unresolved parent or cycle.
			return 0, 0, ErrConflict
		}
	}

	if err := s.cloneVersionVocabulary(

		ctx,

		tx,

		source,

		target,

		nodeIDMap,
	); err != nil {

		return 0, 0, err

	}

	return len(sourceFacets), len(sourceNodes), nil
}

// InspectDraft returns a consistent snapshot of a DRAFT taxonomy version.
// The version row is locked while its facet/node content is read because all
// taxonomy mutation paths lock that same version before editing.
func (s *Service) InspectDraft(
	ctx context.Context,
	versionID string,
) (DraftInspection, error) {
	if s == nil || s.repo == nil || versionID == "" {
		return DraftInspection{}, ErrInvalidArgument
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return DraftInspection{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	version, err := s.repo.GetVersionByIDTx(
		ctx,
		tx,
		versionID,
		true,
	)
	if err != nil {
		return DraftInspection{}, err
	}

	if version.Status != VersionStatusDraft {
		return DraftInspection{}, ErrNotDraft
	}

	facets, err := s.repo.ListFacetsTx(ctx, tx, version.ID)
	if err != nil {
		return DraftInspection{}, err
	}

	nodes, err := s.repo.ListNodesTx(ctx, tx, version.ID)
	if err != nil {
		return DraftInspection{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return DraftInspection{}, err
	}

	return DraftInspection{
		Version: version,
		Facets:  facets,
		Nodes:   nodes,
	}, nil
}

// CompareVersions compares canonical meaning rather than database row IDs.
//
// CloneActiveVersion deliberately gives cloned facets/nodes new identifiers,
// so comparison identity is:
//
//	facet: facet code
//	node:  facet code + canonical node code
//
// Parent relationships are compared by the parent's canonical identity.
func (s *Service) CompareVersions(
	ctx context.Context,
	fromVersionID string,
	toVersionID string,
) (VersionComparison, error) {
	if s == nil ||
		s.repo == nil ||
		fromVersionID == "" ||
		toVersionID == "" {
		return VersionComparison{}, ErrInvalidArgument
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return VersionComparison{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	fromVersion, toVersion, err := s.lockComparisonVersions(
		ctx,
		tx,
		fromVersionID,
		toVersionID,
	)
	if err != nil {
		return VersionComparison{}, err
	}

	if fromVersion.NamespaceID != toVersion.NamespaceID {
		return VersionComparison{}, ErrConflict
	}

	fromFacets, err := s.repo.ListFacetsTx(ctx, tx, fromVersion.ID)
	if err != nil {
		return VersionComparison{}, err
	}

	toFacets, err := s.repo.ListFacetsTx(ctx, tx, toVersion.ID)
	if err != nil {
		return VersionComparison{}, err
	}

	fromNodes, err := s.repo.ListNodesTx(ctx, tx, fromVersion.ID)
	if err != nil {
		return VersionComparison{}, err
	}

	toNodes, err := s.repo.ListNodesTx(ctx, tx, toVersion.ID)
	if err != nil {
		return VersionComparison{}, err
	}

	namespace, err := s.repo.GetNamespaceByID(ctx, fromVersion.NamespaceID)
	if err != nil {
		return VersionComparison{}, err
	}

	comparison := VersionComparison{
		NamespaceCode: namespace.Code,
		FromVersion:   fromVersion,
		ToVersion:     toVersion,
	}

	fromFacetByCode := make(map[string]Facet, len(fromFacets))
	toFacetByCode := make(map[string]Facet, len(toFacets))

	for _, facet := range fromFacets {
		fromFacetByCode[facet.Code] = facet
	}

	for _, facet := range toFacets {
		toFacetByCode[facet.Code] = facet
	}

	// Preserve deterministic repository ordering.
	for _, facet := range toFacets {
		before, exists := fromFacetByCode[facet.Code]
		if !exists {
			comparison.AddedFacets = append(comparison.AddedFacets, facet)
			continue
		}

		if !facetContentEqual(before, facet) {
			comparison.ModifiedFacets = append(
				comparison.ModifiedFacets,
				FacetChange{
					Before: before,
					After:  facet,
				},
			)
		}
	}

	for _, facet := range fromFacets {
		if _, exists := toFacetByCode[facet.Code]; !exists {
			comparison.RemovedFacets = append(
				comparison.RemovedFacets,
				facet,
			)
		}
	}

	fromNodeIndex, err := buildNodeComparisonIndex(
		fromFacets,
		fromNodes,
	)
	if err != nil {
		return VersionComparison{}, err
	}

	toNodeIndex, err := buildNodeComparisonIndex(
		toFacets,
		toNodes,
	)
	if err != nil {
		return VersionComparison{}, err
	}

	for _, item := range toNodeIndex.Ordered {
		before, exists := fromNodeIndex.ByKey[item.Key]
		if !exists {
			comparison.AddedNodes = append(
				comparison.AddedNodes,
				item.Node,
			)
			continue
		}

		if !nodeContentEqual(before, item) {
			comparison.ModifiedNodes = append(
				comparison.ModifiedNodes,
				NodeChange{
					Before: before.Node,
					After:  item.Node,
				},
			)
		}
	}

	for _, item := range fromNodeIndex.Ordered {
		if _, exists := toNodeIndex.ByKey[item.Key]; !exists {
			comparison.RemovedNodes = append(
				comparison.RemovedNodes,
				item.Node,
			)
		}
	}

	if err = tx.Commit(ctx); err != nil {
		return VersionComparison{}, err
	}

	return comparison, nil
}

func (s *Service) lockComparisonVersions(
	ctx context.Context,
	tx pgx.Tx,
	fromID string,
	toID string,
) (Version, Version, error) {
	ids := []string{fromID, toID}

	if fromID != toID {
		sort.Strings(ids)
	}

	versions := make(map[string]Version, len(ids))

	for _, id := range ids {
		if _, exists := versions[id]; exists {
			continue
		}

		version, err := s.repo.GetVersionByIDTx(
			ctx,
			tx,
			id,
			true,
		)
		if err != nil {
			return Version{}, Version{}, err
		}

		versions[id] = version
	}

	return versions[fromID], versions[toID], nil
}

type nodeComparisonItem struct {
	Key       string
	ParentKey string
	Node      Node
}

type nodeComparisonIndex struct {
	ByKey   map[string]nodeComparisonItem
	Ordered []nodeComparisonItem
}

func buildNodeComparisonIndex(
	facets []Facet,
	nodes []Node,
) (nodeComparisonIndex, error) {
	facetCodeByID := make(map[string]string, len(facets))

	for _, facet := range facets {
		facetCodeByID[facet.ID] = facet.Code
	}

	nodeKeyByID := make(map[string]string, len(nodes))

	for _, node := range nodes {
		facetCode, exists := facetCodeByID[node.FacetID]
		if !exists {
			return nodeComparisonIndex{}, ErrConflict
		}

		nodeKeyByID[node.ID] = taxonomyNodeKey(
			facetCode,
			node.CanonicalCode,
		)
	}

	result := nodeComparisonIndex{
		ByKey:   make(map[string]nodeComparisonItem, len(nodes)),
		Ordered: make([]nodeComparisonItem, 0, len(nodes)),
	}

	for _, node := range nodes {
		key := nodeKeyByID[node.ID]

		item := nodeComparisonItem{
			Key:  key,
			Node: node,
		}

		if node.ParentID != nil {
			parentKey, exists := nodeKeyByID[*node.ParentID]
			if !exists {
				return nodeComparisonIndex{}, ErrConflict
			}

			item.ParentKey = parentKey
		}

		result.ByKey[key] = item
		result.Ordered = append(result.Ordered, item)
	}

	return result, nil
}

func taxonomyNodeKey(
	facetCode string,
	nodeCode string,
) string {
	return facetCode + "\x00" + nodeCode
}

func facetContentEqual(
	left Facet,
	right Facet,
) bool {
	return left.Code == right.Code &&
		left.DisplayName == right.DisplayName &&
		left.Description == right.Description &&
		left.StructureType == right.StructureType &&
		left.Position == right.Position &&
		left.Status == right.Status &&
		jsonEquivalent(left.Metadata, right.Metadata)
}

func nodeContentEqual(
	left nodeComparisonItem,
	right nodeComparisonItem,
) bool {
	return left.Node.CanonicalCode == right.Node.CanonicalCode &&
		left.Node.DisplayName == right.Node.DisplayName &&
		left.Node.Description == right.Node.Description &&
		left.ParentKey == right.ParentKey &&
		left.Node.Position == right.Node.Position &&
		left.Node.Status == right.Node.Status &&
		jsonEquivalent(left.Node.Metadata, right.Node.Metadata)
}

func jsonEquivalent(
	left json.RawMessage,
	right json.RawMessage,
) bool {
	var leftValue any
	var rightValue any

	if len(left) == 0 {
		left = json.RawMessage(`{}`)
	}

	if len(right) == 0 {
		right = json.RawMessage(`{}`)
	}

	if err := json.Unmarshal(left, &leftValue); err != nil {
		return false
	}

	if err := json.Unmarshal(right, &rightValue); err != nil {
		return false
	}

	return reflect.DeepEqual(leftValue, rightValue)
}

func (s *Service) replayFacetCommand(
	ctx context.Context,
	tx pgx.Tx,
	namespaceID string,
	commandRef string,
	commandType string,
	fingerprint string,
) (Facet, error) {
	record, err := s.repo.GetCommandTx(
		ctx,
		tx,
		namespaceID,
		commandRef,
	)
	if err != nil {
		return Facet{}, err
	}

	if record.CommandType != commandType ||
		record.RequestFingerprint != fingerprint ||
		record.ResultType != resultTypeFacet {
		return Facet{}, ErrIdempotency
	}

	facet, err := s.repo.GetFacetByIDTx(
		ctx,
		tx,
		record.ResultRef,
		false,
	)
	if err != nil {
		return Facet{}, err
	}

	if facet.NamespaceID != namespaceID ||
		facet.Revision < record.ResultRevision {
		return Facet{}, ErrConflict
	}

	return facet, nil
}

		}

			parent.VersionID != version.ID ||
			parent.FacetID != facet.ID {
			return Node{}, ErrConflict
		}

			return Node{}, ErrConflict
		}
	}

	node, err := s.repo.InsertNode(
		ctx,
		tx,
		Node{
			ID:            "ctn_" + uuid.NewString(),

			VersionID:     version.ID,

		},

	if err != nil {
		return Node{}, err
	}

	if err = s.repo.InsertChangeHistory(
		ctx,
		tx,
		ChangeHistoryEntry{
			ID:            "ctchg_" + uuid.NewString(),
			NamespaceID:   version.NamespaceID,
			VersionID:     version.ID,
			ActorRef:      input.Actor.ActorRef,
			CommandRef:    input.Actor.CommandRef,
			ChangeType:    changeNodeCreated,
			AggregateType: aggregateNode,
			AggregateRef:  node.ID,
			BeforeState:   nil,
			AfterState:    node,
			CorrelationID: input.Actor.CorrelationID,
			CausationID:   input.Actor.CausationID,
		},

		return Node{}, err
	}

	if s.publisher != nil {
		if err = s.publisher.Publish(
			ctx,
			tx,
			eventNodeCreated,
			aggregateNode,
			node.ID,
			node.Revision,
			"",
			input.Actor.CorrelationID,
			input.Actor.CausationID,
			map[string]any{
				"namespace_id":   node.NamespaceID,
				"version_id":     node.VersionID,
				"facet_id":       node.FacetID,
				"node_id":        node.ID,
				"canonical_code": node.CanonicalCode,
				"display_name":   node.DisplayName,
				"parent_id":      node.ParentID,
				"status":         node.Status,
				"revision":       node.Revision,
			},
		); err != nil {
			return Node{}, err
		}
	}

	if err = s.repo.InsertCommand(
		ctx,
		tx,
		CommandRecord{
			ID:                 "ctcmd_" + uuid.NewString(),
			NamespaceID:        version.NamespaceID,
			CommandRef:         input.Actor.CommandRef,
			CommandType:        commandCreateNode,
			RequestFingerprint: fingerprint,
			ResultType:         resultTypeNode,
			ResultRef:          node.ID,
			ResultRevision:     node.Revision,
		},

		return Node{}, err
	}

		return Node{}, err
	}

	return node, nil

	ctx context.Context,
	tx pgx.Tx,
	namespaceID string,
	commandRef string,
	commandType string,
	fingerprint string,

	record, err := s.repo.GetCommandTx(
		ctx,
		tx,
		namespaceID,
		commandRef,
	)
	if err != nil {
		return Node{}, err
	}

	if record.CommandType != commandType ||
		record.RequestFingerprint != fingerprint ||
		record.ResultType != resultTypeNode {
		return Node{}, ErrIdempotency
	}

	node, err := s.repo.GetNodeByIDTx(
		ctx,
		tx,
		record.ResultRef,
		false,
	)
	if err != nil {
		return Node{}, err
	}

	if node.NamespaceID != namespaceID ||
		node.Revision < record.ResultRevision {
		return Node{}, ErrConflict
	}

	return node, nil
}

// UpdateNode changes mutable canonical-node content while the owning taxonomy
// version remains DRAFT.
func (s *Service) UpdateNode(
	ctx context.Context,
	input UpdateNodeInput,
) (Node, error) {
	if s == nil ||
		s.repo == nil ||
		input.NodeID == "" ||
		input.ExpectedRevision <= 0 ||
		input.Actor.ActorRef == "" ||
		input.Actor.CommandRef == "" {
		return Node{}, ErrInvalidArgument
	}

	input.CanonicalCode = strings.ToUpper(strings.TrimSpace(input.CanonicalCode))
	input.DisplayName = strings.TrimSpace(input.DisplayName)
	input.Description = strings.TrimSpace(input.Description)

	if input.CanonicalCode == "" ||
		input.DisplayName == "" ||
		input.Position < 0 {
		return Node{}, ErrInvalidArgument
	}

	if input.ParentID != nil {
		parent := strings.TrimSpace(*input.ParentID)

		if parent == "" {
			input.ParentID = nil
		} else {
			input.ParentID = &parent
		}
	}

	if input.ParentID != nil && *input.ParentID == input.NodeID {
		return Node{}, ErrInvalidArgument
	}

	metadata, err := normalizeMetadata(input.Metadata)
	if err != nil {
		return Node{}, err
	}
	input.Metadata = json.RawMessage(metadata)

	fingerprint, err := requestFingerprint(struct {
		NodeID           string          `json:"node_id"`
		CanonicalCode    string          `json:"canonical_code"`
		DisplayName      string          `json:"display_name"`
		Description      string          `json:"description"`
		ParentID         *string         `json:"parent_id"`
		Position         int             `json:"position"`
		Metadata         json.RawMessage `json:"metadata"`
		ExpectedRevision int64           `json:"expected_revision"`
	}{
		NodeID:           input.NodeID,
		CanonicalCode:    input.CanonicalCode,
		DisplayName:      input.DisplayName,
		Description:      input.Description,
		ParentID:         input.ParentID,
		Position:         input.Position,
		Metadata:         input.Metadata,
		ExpectedRevision: input.ExpectedRevision,
	})
	if err != nil {
		return Node{}, err
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return Node{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	version, facet, current, err := s.loadDraftNodeForMutation(
		ctx,
		tx,
		input.NodeID,
	)
	if err != nil {
		return Node{}, err
	}

	replayed, err := s.replayNodeCommand(
		ctx,
		tx,
		version.NamespaceID,
		input.Actor.CommandRef,
		commandUpdateNode,
		fingerprint,
	)
	if err == nil {
		if err = tx.Commit(ctx); err != nil {
			return Node{}, err
		}
		return replayed, nil
	}
	if !errors.Is(err, ErrNotFound) {
		return Node{}, err
	}

	if current.Revision != input.ExpectedRevision {
		return Node{}, ErrVersionMismatch
	}

	if current.Status != RecordStatusActive ||
		facet.Status != RecordStatusActive {
		return Node{}, ErrConflict
	}

	if err = s.validateNodeParent(
		ctx,
		tx,
		version,
		facet,
		current.ID,
		input.ParentID,
	); err != nil {
		return Node{}, err
	}

	next := current
	next.CanonicalCode = input.CanonicalCode
	next.DisplayName = input.DisplayName
	next.Description = input.Description
	next.ParentID = input.ParentID
	next.Position = input.Position
	next.Metadata = input.Metadata

	updated, err := s.repo.UpdateNode(
		ctx,
		tx,
		next,
		input.ExpectedRevision,
	)

		return Node{}, err

	if err = s.recordNodeMutation(
		ctx,

		current,
		updated,
		commandUpdateNode,
		fingerprint,
		changeNodeUpdated,
		eventNodeUpdated,
	); err != nil {
		return Node{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return Node{}, err
	}

	return updated, nil

func (s *Service) changeNodeStatus(
	ctx context.Context,
	input NodeStatusInput,
	targetStatus RecordStatus,
	commandType string,
	changeType string,
	eventType string,
) (Node, error) {
	if s == nil ||
		s.repo == nil ||
		input.NodeID == "" ||
		input.ExpectedRevision <= 0 ||
		input.Actor.ActorRef == "" ||
		input.Actor.CommandRef == "" {
		return Node{}, ErrInvalidArgument
	}

	fingerprint, err := requestFingerprint(struct {
		NodeID           string       `json:"node_id"`
		TargetStatus     RecordStatus `json:"target_status"`
		ExpectedRevision int64        `json:"expected_revision"`
	}{
		NodeID:           input.NodeID,
		TargetStatus:     targetStatus,
		ExpectedRevision: input.ExpectedRevision,
	})
	if err != nil {
		return Node{}, err
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return Node{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	version, facet, current, err := s.loadDraftNodeForMutation(
		ctx,
		tx,
		input.NodeID,
	)
	if err != nil {
		return Node{}, err
	}

	replayed, err := s.replayNodeCommand(
		ctx,
		tx,
		version.NamespaceID,
		input.Actor.CommandRef,
		commandType,
		fingerprint,
	)
	if err == nil {
		if err = tx.Commit(ctx); err != nil {
			return Node{}, err
		}
		return replayed, nil
	}
	if !errors.Is(err, ErrNotFound) {
		return Node{}, err
	}

	if current.Revision != input.ExpectedRevision {
		return Node{}, ErrVersionMismatch
	}

	if current.Status == targetStatus {
		return Node{}, ErrConflict
	}

	switch targetStatus {
	case RecordStatusArchived:
		hasChildren, checkErr := s.repo.HasActiveChildrenTx(
			ctx,
			tx,
			current.ID,
		)
		if checkErr != nil {
			return Node{}, checkErr
		}

		if hasChildren {
			return Node{}, ErrConflict
		}

	case RecordStatusActive:
		if facet.Status != RecordStatusActive {
			return Node{}, ErrConflict
		}

		if err = s.validateNodeParent(
			ctx,
			tx,
			version,
			facet,
			current.ID,
			current.ParentID,
		); err != nil {
			return Node{}, err
		}

	default:
		return Node{}, ErrInvalidArgument
	}

	next := current
	next.Status = targetStatus

	updated, err := s.repo.UpdateNode(
		ctx,
		tx,
		next,
		input.ExpectedRevision,
	)
	if err != nil {
		return Node{}, err
	}

	if err = s.recordNodeMutation(
		ctx,
		tx,
		input.Actor,
		version,
		current,
		updated,
		commandType,
		fingerprint,
		changeType,
		eventType,
	); err != nil {
		return Node{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return Node{}, err
	}

	return updated, nil
}

func (s *Service) loadDraftNodeForMutation(
	ctx context.Context,
	tx pgx.Tx,
	nodeID string,
) (Version, Facet, Node, error) {
	initial, err := s.repo.GetNodeByIDTx(
		ctx,
		tx,
		nodeID,
		false,
	)
	if err != nil {
		return Version{}, Facet{}, Node{}, err
	}

	version, err := s.repo.GetVersionByIDTx(
		ctx,
		tx,
		initial.VersionID,
		true,
	)
	if err != nil {
		return Version{}, Facet{}, Node{}, err
	}

	if version.Status != VersionStatusDraft {
		return Version{}, Facet{}, Node{}, ErrNotDraft
	}

	node, err := s.repo.GetNodeByIDTx(
		ctx,
		tx,
		nodeID,
		true,
	)
	if err != nil {
		return Version{}, Facet{}, Node{}, err
	}

	if node.VersionID != version.ID ||
		node.NamespaceID != version.NamespaceID {
		return Version{}, Facet{}, Node{}, ErrConflict
	}

	facet, err := s.repo.GetFacetByIDTx(
		ctx,
		tx,
		node.FacetID,
		false,
	)
	if err != nil {
		return Version{}, Facet{}, Node{}, err
	}

	if facet.NamespaceID != version.NamespaceID ||
		facet.VersionID != version.ID ||
		facet.ID != node.FacetID {
		return Version{}, Facet{}, Node{}, ErrConflict
	}

	return version, facet, node, nil
}

func (s *Service) validateNodeParent(
	ctx context.Context,
	tx pgx.Tx,
	version Version,
	facet Facet,
	nodeID string,
	parentID *string,
) error {
	if facet.StructureType == FacetStructureFlat {
		if parentID != nil {
			return ErrInvalidArgument
		}
		return nil
	}

	if parentID == nil {
		return nil
	}

	if *parentID == nodeID {
		return ErrInvalidArgument
	}

	parent, err := s.repo.GetNodeByIDTx(
		ctx,
		tx,
		*parentID,
		false,
	)
	if err != nil {
		return err
	}

	if parent.NamespaceID != version.NamespaceID ||
		parent.VersionID != version.ID ||
		parent.FacetID != facet.ID {
		return ErrConflict
	}

	if parent.Status != RecordStatusActive {
		return ErrConflict
	}

	// Full ancestry cycle and maximum-depth validation remains enforced by
	// canonical_taxonomy_node_guard in PostgreSQL.
	return nil
}

package taxonomy

import (
	"context"
	"errors"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5"
)

const (
	commandPublishVersion = "PUBLISH_VERSION"
	commandRetireVersion  = "RETIRE_VERSION"

	publicationPublished = "PUBLISHED"
	publicationRetired   = "RETIRED"

	eventVersionPublished = "taxonomy.version.published"
	eventVersionRetired   = "taxonomy.version.retired"
)

type PublishVersionInput struct {
	VersionID        string
	ExpectedRevision int64
	Actor            Actor
}

type RetireVersionInput struct {
	VersionID        string
	ExpectedRevision int64
	Actor            Actor
}

// PublishVersion atomically replaces the currently published taxonomy.
//
// Transactional guarantee:
//
//	validate draft
//	    â†“
//	retire current published version
//	    â†“
//	publish new version
//	    â†“
//	publication history
//	    â†“
//	outbox evidence
//	    â†“
//	idempotency evidence
//	    â†“
//	COMMIT
//
// Any failure rolls back the entire transaction, leaving the previous
// published version active.
func (s *Service) PublishVersion(
	ctx context.Context,
	input PublishVersionInput,
) (Version, error) {
	if s == nil ||
		s.repo == nil ||
		input.VersionID == "" ||
		input.ExpectedRevision <= 0 ||
		input.Actor.ActorRef == "" ||
		input.Actor.CommandRef == "" {
		return Version{}, ErrInvalidArgument
	}

	fingerprint, err := requestFingerprint(struct {
		VersionID        string `json:"version_id"`
		ExpectedRevision int64  `json:"expected_revision"`
	}{
		VersionID:        input.VersionID,
		ExpectedRevision: input.ExpectedRevision,
	})
	if err != nil {
		return Version{}, err
	}

	// The version-to-namespace relationship is immutable in PostgreSQL.
	initial, err := s.repo.GetVersionByID(ctx, input.VersionID)
	if err != nil {
		return Version{}, err
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return Version{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	namespace, err := s.repo.LockNamespaceByIDTx(
		ctx,
		tx,
		initial.NamespaceID,
	)
	if err != nil {
		return Version{}, err
	}

	if namespace.Status != NamespaceStatusActive {
		return Version{}, ErrConflict
	}

	draft, err := s.repo.GetVersionByIDTx(
		ctx,
		tx,
		input.VersionID,
		true,
	)
	if err != nil {
		return Version{}, err
	}

	if draft.NamespaceID != namespace.ID {
		return Version{}, ErrConflict
	}

	// Replay is intentionally checked before lifecycle state validation.
	// A successful publish retried with the same command must return the
	// original published result even though it is no longer DRAFT.
	replayed, err := s.replayVersionCommand(
		ctx,
		tx,
		namespace.ID,
		input.Actor.CommandRef,
		commandPublishVersion,
		fingerprint,
	)
	if err == nil {
		if err = tx.Commit(ctx); err != nil {
			return Version{}, err
		}
		return replayed, nil
	}
	if !errors.Is(err, ErrNotFound) {
		return Version{}, err
	}

	if draft.Status != VersionStatusDraft {
		return Version{}, ErrNotDraft
	}

	if draft.Revision != input.ExpectedRevision {
		return Version{}, ErrVersionMismatch
	}

	if err = s.validateDraftForPublication(
		ctx,
		tx,
		draft,
	); err != nil {
		return Version{}, err
	}

	var previous *Version

	current, currentErr := s.repo.GetPublishedVersionTx(
		ctx,
		tx,
		namespace.ID,
		true,
	)

	switch {
	case currentErr == nil:
		previous = &current

	case errors.Is(currentErr, ErrNotFound):
		previous = nil

	default:
		return Version{}, currentErr
	}

	if previous != nil {
		retired, retireErr := s.repo.RetirePublishedVersion(
			ctx,
			tx,
			previous.ID,
			input.Actor.ActorRef,
			previous.Revision,
		)
		if retireErr != nil {
			return Version{}, retireErr
		}

		if err = s.repo.InsertPublicationHistory(
			ctx,
			tx,
			PublicationHistoryEntry{
				ID:            "ctpub_" + uuid.NewString(),
				NamespaceID:   namespace.ID,
				VersionID:     retired.ID,
				EventType:     publicationRetired,
				ActorRef:      input.Actor.ActorRef,
				CommandRef:    input.Actor.CommandRef,
				CorrelationID: input.Actor.CorrelationID,
				CausationID:   input.Actor.CausationID,
				Metadata: map[string]any{
					"replacement_version_id":     draft.ID,
					"replacement_version_number": draft.VersionNumber,
					"retired_version_number":     retired.VersionNumber,
				},
			},
		); err != nil {
			return Version{}, err
		}

		if err = s.publishLifecycleEvent(
			ctx,
			tx,
			input.Actor,
			eventVersionRetired,
			retired,
			map[string]any{
				"replacement_version_id": draft.ID,
			},
		); err != nil {
			return Version{}, err
		}
	}

	published, err := s.repo.PublishDraftVersion(
		ctx,
		tx,
		draft.ID,
		input.Actor.ActorRef,
		draft.Revision,
	)
	if err != nil {
		return Version{}, err
	}

	var previousID *string
	if previous != nil {
		id := previous.ID
		previousID = &id
	}

	if err = s.repo.InsertPublicationHistory(
		ctx,
		tx,
		PublicationHistoryEntry{
			ID:                "ctpub_" + uuid.NewString(),
			NamespaceID:       namespace.ID,
			VersionID:         published.ID,
			PreviousVersionID: previousID,
			EventType:         publicationPublished,
			ActorRef:          input.Actor.ActorRef,
			CommandRef:        input.Actor.CommandRef,
			CorrelationID:     input.Actor.CorrelationID,
			CausationID:       input.Actor.CausationID,
			Metadata: map[string]any{
				"version_number": published.VersionNumber,
			},
		},
	); err != nil {
		return Version{}, err
	}

	if err = s.publishLifecycleEvent(
		ctx,
		tx,
		input.Actor,
		eventVersionPublished,
		published,
		map[string]any{
			"previous_version_id": previousID,
		},
	); err != nil {
		return Version{}, err
	}

	if err = s.repo.InsertCommand(
		ctx,
		tx,
		CommandRecord{
			ID:                 "ctcmd_" + uuid.NewString(),
			NamespaceID:        namespace.ID,
			CommandRef:         input.Actor.CommandRef,
			CommandType:        commandPublishVersion,
			RequestFingerprint: fingerprint,
			ResultType:         resultTypeVersion,
			ResultRef:          published.ID,
			ResultRevision:     published.Revision,
		},
	); err != nil {
		return Version{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return Version{}, err
	}

	return published, nil
}

// RetireVersion explicitly retires the currently published version without
// replacing it. The namespace may temporarily have no published taxonomy.
func (s *Service) RetireVersion(
	ctx context.Context,
	input RetireVersionInput,
) (Version, error) {
	if s == nil ||
		s.repo == nil ||
		input.VersionID == "" ||
		input.ExpectedRevision <= 0 ||
		input.Actor.ActorRef == "" ||
		input.Actor.CommandRef == "" {
		return Version{}, ErrInvalidArgument
	}

	fingerprint, err := requestFingerprint(struct {
		VersionID        string `json:"version_id"`
		ExpectedRevision int64  `json:"expected_revision"`
	}{
		VersionID:        input.VersionID,
		ExpectedRevision: input.ExpectedRevision,
	})
	if err != nil {
		return Version{}, err
	}

	initial, err := s.repo.GetVersionByID(ctx, input.VersionID)
	if err != nil {
		return Version{}, err
	}

	tx, err := s.repo.Begin(ctx)
	if err != nil {
		return Version{}, err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	namespace, err := s.repo.LockNamespaceByIDTx(
		ctx,
		tx,
		initial.NamespaceID,
	)
	if err != nil {
		return Version{}, err
	}

	version, err := s.repo.GetVersionByIDTx(
		ctx,
		tx,
		input.VersionID,
		true,
	)
	if err != nil {
		return Version{}, err
	}

	replayed, err := s.replayVersionCommand(
		ctx,
		tx,
		namespace.ID,
		input.Actor.CommandRef,
		commandRetireVersion,
		fingerprint,
	)
	if err == nil {
		if err = tx.Commit(ctx); err != nil {
			return Version{}, err
		}
		return replayed, nil
	}
	if !errors.Is(err, ErrNotFound) {
		return Version{}, err
	}

	if version.Status != VersionStatusPublished {
		return Version{}, ErrConflict
	}

	if version.Revision != input.ExpectedRevision {
		return Version{}, ErrVersionMismatch
	}

	retired, err := s.repo.RetirePublishedVersion(
		ctx,
		tx,
		version.ID,
		input.Actor.ActorRef,
		version.Revision,
	)
	if err != nil {
		return Version{}, err
	}

	if err = s.repo.InsertPublicationHistory(
		ctx,
		tx,
		PublicationHistoryEntry{
			ID:            "ctpub_" + uuid.NewString(),
			NamespaceID:   namespace.ID,
			VersionID:     retired.ID,
			EventType:     publicationRetired,
			ActorRef:      input.Actor.ActorRef,
			CommandRef:    input.Actor.CommandRef,
			CorrelationID: input.Actor.CorrelationID,
			CausationID:   input.Actor.CausationID,
			Metadata: map[string]any{
				"version_number": retired.VersionNumber,
				"explicit":       true,
			},
		},
	); err != nil {
		return Version{}, err
	}

	if err = s.publishLifecycleEvent(
		ctx,
		tx,
		input.Actor,
		eventVersionRetired,
		retired,
		map[string]any{
			"explicit": true,
		},
	); err != nil {
		return Version{}, err
	}

	if err = s.repo.InsertCommand(
		ctx,
		tx,
		CommandRecord{
			ID:                 "ctcmd_" + uuid.NewString(),
			NamespaceID:        namespace.ID,
			CommandRef:         input.Actor.CommandRef,
			CommandType:        commandRetireVersion,
			RequestFingerprint: fingerprint,
			ResultType:         resultTypeVersion,
			ResultRef:          retired.ID,
			ResultRevision:     retired.Revision,
		},
	); err != nil {
		return Version{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return Version{}, err
	}

	return retired, nil
}

// validateDraftForPublication checks cross-record invariants immediately
// before lifecycle replacement.
//
// PostgreSQL remains the final authority for FK ownership, cycles, bounded
// hierarchy depth and lifecycle immutability.
func (s *Service) validateDraftForPublication(
	ctx context.Context,
	tx pgx.Tx,
	version Version,
) error {
	if version.Status != VersionStatusDraft {
		return ErrNotDraft
	}

	facets, err := s.repo.ListFacetsTx(ctx, tx, version.ID)
	if err != nil {
		return err
	}

	nodes, err := s.repo.ListNodesTx(ctx, tx, version.ID)
	if err != nil {
		return err
	}

	facetByID := make(map[string]Facet, len(facets))
	for _, facet := range facets {
		if facet.NamespaceID != version.NamespaceID ||
			facet.VersionID != version.ID {
			return ErrConflict
		}

		facetByID[facet.ID] = facet
	}

	nodeByID := make(map[string]Node, len(nodes))
	for _, node := range nodes {
		if node.NamespaceID != version.NamespaceID ||
			node.VersionID != version.ID {
			return ErrConflict
		}

		nodeByID[node.ID] = node
	}

	for _, node := range nodes {
		facet, exists := facetByID[node.FacetID]
		if !exists {
			return ErrConflict
		}

		if facet.StructureType == FacetStructureFlat &&
			node.ParentID != nil {
			return ErrConflict
		}

		if node.Status == RecordStatusActive &&
			facet.Status != RecordStatusActive {
			return ErrConflict
		}

		if node.ParentID == nil {
			continue
		}

		parent, exists := nodeByID[*node.ParentID]
		if !exists {
			return ErrConflict
		}

		if parent.FacetID != node.FacetID ||
			parent.NamespaceID != node.NamespaceID ||
			parent.VersionID != node.VersionID {
			return ErrConflict
		}

		if node.Status == RecordStatusActive &&
			parent.Status != RecordStatusActive {
			return ErrConflict
		}
	}

	return nil
}

func (s *Service) publishLifecycleEvent(
	ctx context.Context,
	tx pgx.Tx,
	actor Actor,
	eventType string,
	version Version,
	extra map[string]any,
) error {
	if s.publisher == nil {
		return nil
	}

	payload := map[string]any{
		"namespace_id":   version.NamespaceID,
		"version_id":     version.ID,
		"version_number": version.VersionNumber,
		"status":         version.Status,
		"revision":       version.Revision,
	}

	for key, value := range extra {
		payload[key] = value
	}

	return s.publisher.Publish(
		ctx,
		tx,
		eventType,
		aggregateVersion,
		version.ID,
		version.Revision,
		"",
		actor.CorrelationID,
		actor.CausationID,
		payload,
	)
}

type CreateMappingInput struct {

	VersionID string
	NodeID    string

	SourceType string
	SourceRef  string

	MappingType MappingType

	ConfidenceBasis MappingConfidenceBasis
	ConfidenceScore *int

	BusinessRef *string
	BranchRef   *string

	Metadata json.RawMessage

	Actor Actor

func deterministicMappingID(
	input CreateMappingInput,
) string {
	raw := strings.Join(
		[]string{
			string(input.NamespaceCode),
			input.VersionID,
			input.NodeID,
			strings.ToUpper(
				strings.TrimSpace(input.SourceType),
			),
			strings.TrimSpace(input.SourceRef),
			string(input.MappingType),
		},
		"|",
	)

	sum := sha256.Sum256([]byte(raw))

	return "ctmap_" +
		hex.EncodeToString(sum[:16])
}

func (s *Service) CreateMapping(
	ctx context.Context,

		strings.TrimSpace(input.Actor.CommandRef) == "" {

	case MappingTypePrimary,
		MappingTypeSecondary,
		MappingTypeDiscovery:

		MappingConfidenceRule,
		MappingConfidenceImport,
		MappingConfidenceSystem:

		(*input.ConfidenceScore < 0 ||
			*input.ConfidenceScore > 10000) {
		return TaxonomyMapping{}, ErrInvalidArgument

            branch_ref,

        `,

		input.ConfidenceScore,
		valueOrEmpty(input.BusinessRef),
		valueOrEmpty(input.BranchRef),

		&mapping.NodeID,
		&mapping.SourceType,

		&mapping.Status,

		&mapping.BusinessRef,
		&mapping.BranchRef,
		&mapping.Metadata,
		&mapping.Revision,
		&mapping.CreatedByActorRef,
		&mapping.UpdatedByActorRef,
		&mapping.CreatedAt,
		&mapping.UpdatedAt,
	)
	if err != nil {
		return TaxonomyMapping{}, err
	}

	if err = tx.Commit(ctx); err != nil {
		return TaxonomyMapping{}, err
	}

	return mapping, nil
}

func (s *Service) ReplaceFacetMappings(
	ctx context.Context,
	input ReplaceFacetMappingsInput,
) (ReplaceFacetMappingsResult, error) {
	if s == nil ||
		s.repo == nil ||
		strings.TrimSpace(input.Actor.ActorRef) == "" ||
		strings.TrimSpace(input.Actor.CommandRef) == "" {
		return ReplaceFacetMappingsResult{},
			ErrInvalidArgument
	}

	normalized, requestHash, err :=
		prepareReplaceFacetMappingsInput(input)
	if err != nil {
		return ReplaceFacetMappingsResult{},
			err
	}

	tx, err := s.repo.pool.BeginTx(
		ctx,
		pgx.TxOptions{
			IsoLevel: pgx.Serializable,
		},
	)
	if err != nil {
		return ReplaceFacetMappingsResult{},
			err
	}

	defer func() {
		_ = tx.Rollback(ctx)
	}()

	// Serialize identical command references.
	if _, err = tx.Exec(
		ctx,
		`
        SELECT pg_advisory_xact_lock(
            hashtext($1)
        )
        `,
		"taxonomy-command|"+
			strings.TrimSpace(
				input.Actor.CommandRef,
			),
	); err != nil {
		return ReplaceFacetMappingsResult{},
			err
	}

	var existingHash string
	var existingSnapshot []byte

	err = tx.QueryRow(
		ctx,
		`
        SELECT
            request_hash,
            result_snapshot
          FROM canonical_taxonomy_mapping_commands
         WHERE command_ref=$1
        `,
		strings.TrimSpace(
			input.Actor.CommandRef,
		),
	).Scan(
		&existingHash,
		&existingSnapshot,
	)

	if err == nil {
		if existingHash != requestHash {
			return ReplaceFacetMappingsResult{},
				fmt.Errorf(
					"%w: command_ref reused with different mapping request",
					ErrInvalidArgument,
				)
		}

		var replay ReplaceFacetMappingsResult

		if err = json.Unmarshal(
			existingSnapshot,
			&replay,
		); err != nil {
			return ReplaceFacetMappingsResult{},
				err
		}

		replay.Replayed = true

		return replay, nil
	}

	if !errors.Is(err, pgx.ErrNoRows) {
		return ReplaceFacetMappingsResult{},
			err
	}

	var namespaceID string
	var versionID string

	err = tx.QueryRow(
		ctx,
		`
        SELECT
            ns.id,
            v.id
          FROM canonical_taxonomy_namespaces ns
          JOIN canonical_taxonomy_versions v
            ON v.namespace_id=ns.id
         WHERE ns.code=$1
           AND ns.status='ACTIVE'
           AND v.status='PUBLISHED'
         ORDER BY v.version_number DESC
         LIMIT 1
        `,
		normalized.NamespaceCode,
	).Scan(
		&namespaceID,
		&versionID,
	)
	if err != nil {
		return ReplaceFacetMappingsResult{},
			err
	}

	var facetID string

	err = tx.QueryRow(
		ctx,
		`
        SELECT id
          FROM canonical_taxonomy_facets
         WHERE namespace_id=$1
           AND version_id=$2
           AND code=$3
        `,
		namespaceID,
		versionID,
		normalized.FacetCode,
	).Scan(
		&facetID,
	)
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return ReplaceFacetMappingsResult{},
				ErrNotFound
		}

		return ReplaceFacetMappingsResult{},
			err
	}

	// Serialize mutation of one source/facet mapping set.
	if _, err = tx.Exec(
		ctx,
		`
        SELECT pg_advisory_xact_lock(
            hashtext($1)
        )
        `,

			[]string{
				"taxonomy-source-facet",

			ctx,
			`
            SELECT id
              FROM canonical_taxonomy_nodes
             WHERE namespace_id=$1
               AND version_id=$2
               AND facet_id=$3
               AND canonical_code=$4
               AND status='ACTIVE'
            `,
			namespaceID,
			versionID,
			facetID,
			candidate.NodeCode,
		).Scan(
			&nodeID,
		)

			if errors.Is(err, pgx.ErrNoRows) {

						"%w: node %s is not active in facet %s",
						ErrNotFound,
						candidate.NodeCode,
						normalized.FacetCode,
					)
			}

			return ReplaceFacetMappingsResult{},
				err
		}

		mappingID := deterministicMappingID(
			CreateMappingInput{
				NamespaceCode: normalized.NamespaceCode,

				VersionID: versionID,
				NodeID:    nodeID,

				SourceType: normalized.SourceType,

				SourceRef: normalized.SourceRef,

				MappingType: candidate.MappingType,
			},
		)

				ID:        mappingID,
				NodeID:    nodeID,
				Candidate: candidate,
			},
		)
	}

	// archive the current active set for this source/facet,
	// then reactivate/upsert exactly the requested set.

           AND version_id=$3

			err

                namespace_id,
                version_id,
                source_type,
                source_ref,
                node_id,
                mapping_type
            )
            DO UPDATE SET
                status='ACTIVE',
                confidence_basis=
                    EXCLUDED.confidence_basis,
                confidence_score=
                    EXCLUDED.confidence_score,
                business_ref=
                    EXCLUDED.business_ref,
                branch_ref=
                    EXCLUDED.branch_ref,
                metadata=
                    EXCLUDED.metadata,
                updated_by_actor_ref=
                    EXCLUDED.updated_by_actor_ref
            `,

			namespaceID,
			versionID,
			facetID,
			item.NodeID,
			normalized.SourceType,
			normalized.SourceRef,
			item.Candidate.MappingType,
			item.Candidate.ConfidenceBasis,
			item.Candidate.ConfidenceScore,
			normalized.BusinessRef,
			normalized.BranchRef,
			item.Candidate.Metadata,
			input.Actor.ActorRef,
		)
		if err != nil {
			return ReplaceFacetMappingsResult{},
				err
		}

		mappingIDs = append(
			mappingIDs,
			item.ID,
		)
	}

	result := ReplaceFacetMappingsResult{
		NamespaceCode: normalized.NamespaceCode,

		NamespaceID: namespaceID,

		VersionID: versionID,

		FacetID: facetID,

		FacetCode: normalized.FacetCode,

		SourceType: normalized.SourceType,

		SourceRef: normalized.SourceRef,

		MappingIDs: mappingIDs,

		Replayed: false,
	}

	snapshot, err := json.Marshal(result)

	}

	_, err = tx.Exec(

		`

            id,
            command_ref,
            actor_ref,
            operation,
            mapping_id,
            request_hash,
            result_snapshot

        VALUES (

            $5
        )
        `,
		semanticObjectID(

				input.Actor.CommandRef,
			),
		),
		strings.TrimSpace(
			input.Actor.CommandRef,
		),
		input.Actor.ActorRef,
		requestHash,
		snapshot,
	)
	if err != nil {
		return ReplaceFacetMappingsResult{},
			err
	}

	if err = tx.Commit(ctx); err != nil {
		return ReplaceFacetMappingsResult{},
			err
	}

	CaseID string `json:"case_id,omitempty"`

	NamespaceCode NamespaceCode `json:"namespace_code,omitempty"`

	SubjectType string `json:"subject_type,omitempty"`
	SubjectRef  string `json:"subject_ref,omitempty"`

	CaseType GovernanceCaseType `json:"case_type,omitempty"`
	Priority GovernancePriority `json:"priority,omitempty"`

	ExpectedRevision int64 `json:"expected_revision,omitempty"`

	Transition GovernanceTransition `json:"transition,omitempty"`

	AssignedActorRef string `json:"assigned_actor_ref,omitempty"`

	ResolutionCode  string `json:"resolution_code,omitempty"`
	ResolutionNotes string `json:"resolution_notes,omitempty"`

	ReasonCode  string `json:"reason_code,omitempty"`
	Description string `json:"description,omitempty"`

	Metadata json.RawMessage `json:"metadata,omitempty"`
}

func normalizeGovernanceMetadata(
	raw json.RawMessage,
) (json.RawMessage, error) {
	if len(raw) == 0 {
		return json.RawMessage(`{}`), nil
	}

	var object map[string]any

	if err := json.Unmarshal(raw, &object); err != nil {
		return nil, ErrInvalidArgument
	}

	return json.Marshal(object)
}

func (s *Service) OpenGovernanceCase(
	ctx context.Context,
	input OpenGovernanceCaseInput,
) (GovernanceCommandResult, error) {
	if s == nil ||
		s.repo == nil ||
		strings.TrimSpace(input.SubjectType) == "" ||
		strings.TrimSpace(input.SubjectRef) == "" ||
		strings.TrimSpace(input.Actor.ActorRef) == "" ||
		strings.TrimSpace(input.Actor.CommandRef) == "" ||
		!validGovernanceCaseType(input.CaseType) ||
		!validGovernancePriority(input.Priority) {
		return GovernanceCommandResult{},
			ErrInvalidArgument
	}

	if input.NamespaceCode != NamespaceFood &&
		input.NamespaceCode != NamespaceShop {
		return GovernanceCommandResult{},
			ErrInvalidArgument
	}

	metadata, err :=
		normalizeGovernanceMetadata(
			input.Metadata,
		)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	subjectType := strings.ToUpper(
		strings.TrimSpace(input.SubjectType),
	)
	subjectRef := strings.TrimSpace(
		input.SubjectRef,
	)

	envelope := governanceCommandEnvelope{
		Operation: "OPEN_CASE",

		NamespaceCode: input.NamespaceCode,

		SubjectType: subjectType,

		SubjectRef: subjectRef,

		CaseType: input.CaseType,

		Priority: input.Priority,

		ReasonCode: valueOrEmpty(input.ReasonCode),

		Description: valueOrEmpty(input.Description),

		Metadata: metadata,
	}

	requestHash, err :=
		governanceRequestHash(envelope)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	tx, err := s.repo.pool.BeginTx(
		ctx,
		pgx.TxOptions{
			IsoLevel: pgx.Serializable,
		},
	)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	defer func() {
		_ = tx.Rollback(ctx)
	}()

	if _, err = tx.Exec(
		ctx,
		`
        SELECT pg_advisory_xact_lock(
            hashtext($1)
        )
        `,
		"taxonomy-governance-command|"+
			strings.TrimSpace(
				input.Actor.CommandRef,
			),
	); err != nil {
		return GovernanceCommandResult{},
			err
	}

	replayed, result, err :=
		readGovernanceCommandReplay(
			ctx,
			tx,
			strings.TrimSpace(
				input.Actor.CommandRef,
			),
			requestHash,
		)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	if replayed {
		result.Replayed = true

		return result, nil
	}

	var namespaceID string
	var versionID string

	err = tx.QueryRow(
		ctx,
		`
        SELECT
            ns.id,
            v.id
          FROM canonical_taxonomy_namespaces ns
          JOIN canonical_taxonomy_versions v
            ON v.namespace_id=ns.id
         WHERE ns.code=$1
           AND ns.status='ACTIVE'
           AND v.status='PUBLISHED'
         ORDER BY v.version_number DESC
         LIMIT 1
        `,
		input.NamespaceCode,
	).Scan(
		&namespaceID,
		&versionID,
	)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	if _, err = tx.Exec(
		ctx,
		`
        SELECT pg_advisory_xact_lock(
            hashtext($1)
        )
        `,
		strings.Join(
			[]string{
				"taxonomy-governance-subject",
				namespaceID,
				versionID,
				subjectType,
				subjectRef,
				string(input.CaseType),
			},
			"|",
		),
	); err != nil {
		return GovernanceCommandResult{},
			err
	}

	caseID := deterministicGovernanceCaseID(
		input.NamespaceCode,
		versionID,
		subjectType,
		subjectRef,
		input.CaseType,
	)

	var c GovernanceCase

	err = tx.QueryRow(
		ctx,
		`
        INSERT INTO canonical_taxonomy_governance_cases (
            id,
            namespace_id,
            version_id,
            subject_type,
            subject_ref,
            case_type,
            status,
            priority,
            reason_code,
            description,
            metadata,
            revision,
            opened_by_actor_ref
        )
        VALUES (
            $1,$2,$3,$4,$5,$6,
            'OPEN',$7,
            NULLIF($8,''),
            NULLIF($9,''),
            $10,1,$11
        )
        RETURNING
            id,
            namespace_id,
            version_id,
            subject_type,
            subject_ref,
            case_type,
            status,
            priority,
            reason_code,
            description,
            resolution_code,
            resolution_notes,
            assigned_actor_ref,
            opened_by_actor_ref,
            resolved_by_actor_ref,
            metadata,
            revision,
            created_at,
            updated_at,
            resolved_at
        `,
		caseID,
		namespaceID,
		versionID,
		subjectType,
		subjectRef,
		input.CaseType,
		input.Priority,
		valueOrEmpty(input.ReasonCode),
		valueOrEmpty(input.Description),
		metadata,
		input.Actor.ActorRef,
	).Scan(
		&c.ID,
		&c.NamespaceID,
		&c.VersionID,
		&c.SubjectType,
		&c.SubjectRef,
		&c.CaseType,
		&c.Status,
		&c.Priority,
		&c.ReasonCode,
		&c.Description,
		&c.ResolutionCode,
		&c.ResolutionNotes,
		&c.AssignedActorRef,
		&c.OpenedByActorRef,
		&c.ResolvedByActorRef,
		&c.Metadata,
		&c.Revision,
		&c.CreatedAt,
		&c.UpdatedAt,
		&c.ResolvedAt,
	)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	result = GovernanceCommandResult{
		Case:     c,
		Replayed: false,
	}

	if err = insertGovernanceCommand(
		ctx,
		tx,
		input.Actor,
		"OPEN_CASE",
		caseID,
		requestHash,
		result,
	); err != nil {
		return GovernanceCommandResult{},
			err
	}

	if err = tx.Commit(ctx); err != nil {
		return GovernanceCommandResult{},
			err
	}

	return result, nil
}

func (s *Service) TransitionGovernanceCase(
	ctx context.Context,
	input TransitionGovernanceCaseInput,
) (GovernanceCommandResult, error) {
	if s == nil ||
		s.repo == nil ||
		strings.TrimSpace(input.CaseID) == "" ||
		input.ExpectedRevision < 1 ||
		strings.TrimSpace(input.Actor.ActorRef) == "" ||
		strings.TrimSpace(input.Actor.CommandRef) == "" ||
		!validGovernanceTransition(input.Transition) {
		return GovernanceCommandResult{},
			ErrInvalidArgument
	}

	metadata, err :=
		normalizeGovernanceMetadata(
			input.Metadata,
		)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	envelope := governanceCommandEnvelope{
		Operation: string(input.Transition),

		CaseID: strings.TrimSpace(input.CaseID),

		ExpectedRevision: input.ExpectedRevision,

		Transition: input.Transition,

		AssignedActorRef: valueOrEmpty(
			input.AssignedActorRef,
		),

		ResolutionCode: valueOrEmpty(
			input.ResolutionCode,
		),

		ResolutionNotes: valueOrEmpty(
			input.ResolutionNotes,
		),

		Metadata: metadata,
	}

	requestHash, err :=
		governanceRequestHash(envelope)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	tx, err := s.repo.pool.BeginTx(
		ctx,
		pgx.TxOptions{
			IsoLevel: pgx.Serializable,
		},
	)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	defer func() {
		_ = tx.Rollback(ctx)
	}()

	if _, err = tx.Exec(
		ctx,
		`
        SELECT pg_advisory_xact_lock(
            hashtext($1)
        )
        `,
		"taxonomy-governance-command|"+
			strings.TrimSpace(
				input.Actor.CommandRef,
			),
	); err != nil {
		return GovernanceCommandResult{},
			err
	}

	replayed, result, err :=
		readGovernanceCommandReplay(
			ctx,
			tx,
			strings.TrimSpace(
				input.Actor.CommandRef,
			),
			requestHash,
		)
	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	if replayed {
		result.Replayed = true

		return result, nil
	}

	if _, err = tx.Exec(
		ctx,
		`
        SELECT pg_advisory_xact_lock(
            hashtext($1)
        )
        `,
		"taxonomy-governance-case|"+
			strings.TrimSpace(
				input.CaseID,
			),
	); err != nil {
		return GovernanceCommandResult{},
			err
	}

	var nextStatus GovernanceStatus
	var assignedActor string
	var resolutionCode string
	var resolutionNotes string
	var resolvedActor string

	switch input.Transition {
	case GovernanceTransitionStartReview:
		nextStatus =
			GovernanceStatusInReview

		assignedActor =
			valueOrEmpty(
				input.AssignedActorRef,
			)

		if assignedActor == "" {
			assignedActor =
				input.Actor.ActorRef
		}

	case GovernanceTransitionAssign:
		assignedActor =
			valueOrEmpty(
				input.AssignedActorRef,
			)

		if assignedActor == "" {
			return GovernanceCommandResult{},
				ErrInvalidArgument
		}

	case GovernanceTransitionResolve:
		nextStatus =
			GovernanceStatusResolved

		resolutionCode =
			strings.ToUpper(
				valueOrEmpty(
					input.ResolutionCode,
				),
			)

		if resolutionCode == "" {
			return GovernanceCommandResult{},
				ErrInvalidArgument
		}

		resolutionNotes =
			valueOrEmpty(
				input.ResolutionNotes,
			)

		resolvedActor =
			input.Actor.ActorRef

	case GovernanceTransitionReject:
		nextStatus =
			GovernanceStatusRejected

		resolutionCode =
			strings.ToUpper(
				valueOrEmpty(
					input.ResolutionCode,
				),
			)

		if resolutionCode == "" {
			return GovernanceCommandResult{},
				ErrInvalidArgument
		}

		resolutionNotes =
			valueOrEmpty(
				input.ResolutionNotes,
			)

		resolvedActor =
			input.Actor.ActorRef

	case GovernanceTransitionCancel:
		nextStatus =
			GovernanceStatusCancelled

		resolutionNotes =
			valueOrEmpty(
				input.ResolutionNotes,
			)

		resolvedActor =
			input.Actor.ActorRef
	}

	var c GovernanceCase

	if input.Transition ==
		GovernanceTransitionAssign {

		err = tx.QueryRow(
			ctx,
			`
            UPDATE canonical_taxonomy_governance_cases
               SET assigned_actor_ref=$1
             WHERE id=$2
               AND revision=$3
               AND status IN (
                   'OPEN',
                   'IN_REVIEW'
               )
            RETURNING
                id,
                namespace_id,
                version_id,
                subject_type,
                subject_ref,
                case_type,
                status,
                priority,
                reason_code,
                description,
                resolution_code,
                resolution_notes,
                assigned_actor_ref,
                opened_by_actor_ref,
                resolved_by_actor_ref,
                metadata,
                revision,
                created_at,
                updated_at,
                resolved_at
            `,
			assignedActor,
			input.CaseID,
			input.ExpectedRevision,
		).Scan(
			&c.ID,
			&c.NamespaceID,
			&c.VersionID,
			&c.SubjectType,
			&c.SubjectRef,
			&c.CaseType,
			&c.Status,
			&c.Priority,
			&c.ReasonCode,
			&c.Description,
			&c.ResolutionCode,
			&c.ResolutionNotes,
			&c.AssignedActorRef,
			&c.OpenedByActorRef,
			&c.ResolvedByActorRef,
			&c.Metadata,
			&c.Revision,
			&c.CreatedAt,
			&c.UpdatedAt,
			&c.ResolvedAt,
		)
	} else {
		err = tx.QueryRow(
			ctx,
			`
            UPDATE canonical_taxonomy_governance_cases
               SET status=$1,
                   assigned_actor_ref=
                       CASE
                           WHEN NULLIF($2,'') IS NULL
                               THEN assigned_actor_ref
                           ELSE $2
                       END,
                   resolution_code=
                       NULLIF($3,''),
                   resolution_notes=
                       NULLIF($4,''),
                   resolved_by_actor_ref=
                       NULLIF($5,''),
                   metadata=
                       CASE
                           WHEN $6::jsonb='{}'::jsonb
                               THEN metadata
                           ELSE $6::jsonb
                       END
             WHERE id=$7
               AND revision=$8
               AND status IN (
                   'OPEN',
                   'IN_REVIEW'
               )
            RETURNING
                id,
                namespace_id,
                version_id,
                subject_type,
                subject_ref,
                case_type,
                status,
                priority,
                reason_code,
                description,
                resolution_code,
                resolution_notes,
                assigned_actor_ref,
                opened_by_actor_ref,
                resolved_by_actor_ref,
                metadata,
                revision,
                created_at,
                updated_at,
                resolved_at
            `,
			nextStatus,
			assignedActor,
			resolutionCode,
			resolutionNotes,
			resolvedActor,
			metadata,
			input.CaseID,
			input.ExpectedRevision,
		).Scan(
			&c.ID,
			&c.NamespaceID,
			&c.VersionID,
			&c.SubjectType,
			&c.SubjectRef,
			&c.CaseType,
			&c.Status,
			&c.Priority,
			&c.ReasonCode,
			&c.Description,
			&c.ResolutionCode,
			&c.ResolutionNotes,
			&c.AssignedActorRef,
			&c.OpenedByActorRef,
			&c.ResolvedByActorRef,
			&c.Metadata,
			&c.Revision,
			&c.CreatedAt,
			&c.UpdatedAt,
			&c.ResolvedAt,
		)
	}

	if errors.Is(err, pgx.ErrNoRows) {
		return GovernanceCommandResult{},
			fmt.Errorf(
				"%w: governance case/revision/state conflict",
				ErrInvalidArgument,
			)
	}

	if err != nil {
		return GovernanceCommandResult{},
			err
	}

	result = GovernanceCommandResult{
		Case:     c,
		Replayed: false,
	}

	if err = insertGovernanceCommand(
		ctx,
		tx,
		input.Actor,
		string(input.Transition),
		input.CaseID,
		requestHash,
		result,
	); err != nil {
		return GovernanceCommandResult{},
			err
	}

	if err = tx.Commit(ctx); err != nil {
		return GovernanceCommandResult{},
			err
	}

	return result, nil
}

func readGovernanceCommandReplay(
	ctx context.Context,
	tx pgx.Tx,
	commandRef string,
	requestHash string,
) (
	bool,
	GovernanceCommandResult,
	error,
) {
	var existingHash string
	var snapshot []byte

	err := tx.QueryRow(
		ctx,
		`
        SELECT
            request_hash,
            result_snapshot
          FROM canonical_taxonomy_governance_commands
         WHERE command_ref=$1
        `,
		commandRef,
	).Scan(
		&existingHash,
		&snapshot,
	)

	if errors.Is(err, pgx.ErrNoRows) {
		return false,
			GovernanceCommandResult{},
			nil
	}

	if err != nil {
		return false,
			GovernanceCommandResult{},
			err
	}

	if existingHash != requestHash {
		return false,
			GovernanceCommandResult{},
			fmt.Errorf(
				"%w: governance command_ref reused with different request",
				ErrInvalidArgument,
			)
	}

	var result GovernanceCommandResult

	if err = json.Unmarshal(
		snapshot,
		&result,
	); err != nil {
		return false,
			GovernanceCommandResult{},
			err
	}

	return true, result, nil
}

func insertGovernanceCommand(
	ctx context.Context,
	tx pgx.Tx,
	actor Actor,
	operation string,
	caseID string,
	requestHash string,
	result GovernanceCommandResult,
) error {
	snapshot, err :=
		json.Marshal(result)
	if err != nil {
		return err
	}

	_, err = tx.Exec(
		ctx,
		`
        INSERT INTO canonical_taxonomy_governance_commands (
            id,
            command_ref,
            actor_ref,
            operation,
            case_id,
            request_hash,
            result_snapshot
        )
        VALUES (
            $1,$2,$3,$4,$5,$6,$7
        )
        `,
		semanticObjectID(
			"ctgovcmd_",
			strings.TrimSpace(
				actor.CommandRef,
			),
		),
		strings.TrimSpace(
			actor.CommandRef,
		),
		actor.ActorRef,
		operation,
		caseID,
		requestHash,
		snapshot,
	)

	return err
}

func normalizeMetadata(metadata json.RawMessage) (string, error) {
	if len(metadata) == 0 {
		return `{}`, nil
	}

	if !json.Valid(metadata) {
		return "", ErrInvalidArgument
	}

	var object map[string]any
	if err := json.Unmarshal(metadata, &object); err != nil || object == nil {
		return "", ErrInvalidArgument
	}

	return string(metadata), nil
}

	return ErrVersionMismatch
}

-- ============================================================================
-- 3. Version lifecycle guard
-- ============================================================================
--
-- Drafts may be edited.
-- Published versions may only transition to RETIRED.
-- Retired versions are immutable.
-- Historical versions cannot be deleted.

CREATE FUNCTION canonical_taxonomy_version_guard()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        IF NEW.status <> 'DRAFT' THEN
            RAISE EXCEPTION
                'canonical taxonomy versions must be created as DRAFT';
        END IF;

        RETURN NEW;
    END IF;

    IF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION
            'canonical taxonomy version history is immutable';
    END IF;

    IF NEW.namespace_id <> OLD.namespace_id
       OR NEW.version_number <> OLD.version_number
       OR NEW.source_version_id IS DISTINCT FROM OLD.source_version_id
       OR NEW.created_by_actor_ref <> OLD.created_by_actor_ref
       OR NEW.created_at <> OLD.created_at THEN
        RAISE EXCEPTION
            'canonical taxonomy version identity is immutable';
    END IF;

    IF OLD.status = 'RETIRED' THEN
        RAISE EXCEPTION
            'retired canonical taxonomy versions are immutable';
    END IF;

    IF OLD.status = 'PUBLISHED' THEN
        IF NEW.status <> 'RETIRED' THEN
            RAISE EXCEPTION
                'published canonical taxonomy versions may only transition to RETIRED';
        END IF;

        IF NEW.published_at IS DISTINCT FROM OLD.published_at
           OR NEW.published_by_actor_ref IS DISTINCT FROM OLD.published_by_actor_ref THEN
            RAISE EXCEPTION
                'publication evidence is immutable';
        END IF;

        RETURN NEW;
    END IF;

    -- OLD is DRAFT here.
    IF NEW.status NOT IN ('DRAFT','PUBLISHED') THEN
        RAISE EXCEPTION
            'draft canonical taxonomy versions may only remain DRAFT or become PUBLISHED';
    END IF;

    RETURN NEW;
END;
$$;

-- ============================================================================
-- 7. Node hierarchy + draft guard
-- ============================================================================
--
-- Maximum hierarchy depth is 12 edges from a node to its root.
-- FLAT facets cannot have parents.
-- Parent composite FK guarantees same namespace/version/facet.
-- Recursive validation rejects cycles.

CREATE FUNCTION canonical_taxonomy_node_guard()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    owner_namespace text;
    owner_version   text;
    version_status  text;
    facet_structure text;
    parent_exists   boolean;
    cycle_found     boolean;
    ancestor_depth  integer;
BEGIN
    IF TG_OP = 'DELETE' THEN
        owner_namespace := OLD.namespace_id;
        owner_version := OLD.version_id;
    ELSE
        owner_namespace := NEW.namespace_id;
        owner_version := NEW.version_id;
    END IF;

    IF TG_OP = 'UPDATE'
       AND (
            NEW.namespace_id <> OLD.namespace_id
            OR NEW.version_id <> OLD.version_id
            OR NEW.facet_id <> OLD.facet_id
       ) THEN
        RAISE EXCEPTION
            'canonical taxonomy nodes cannot move between namespaces, versions or facets';
    END IF;

    SELECT status
      INTO version_status
      FROM canonical_taxonomy_versions
     WHERE id = owner_version
       AND namespace_id = owner_namespace;

    IF version_status IS DISTINCT FROM 'DRAFT' THEN
        RAISE EXCEPTION
            'canonical taxonomy node content is mutable only in DRAFT versions';
    END IF;

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;

    SELECT structure_type
      INTO facet_structure
      FROM canonical_taxonomy_facets
     WHERE id = NEW.facet_id
       AND namespace_id = NEW.namespace_id
       AND version_id = NEW.version_id;

    IF facet_structure IS NULL THEN
        RAISE EXCEPTION
            'canonical taxonomy facet does not exist in this namespace/version';
    END IF;

    IF facet_structure = 'FLAT' AND NEW.parent_id IS NOT NULL THEN
        RAISE EXCEPTION
            'FLAT canonical taxonomy facets cannot contain parent relationships';
    END IF;

    IF NEW.parent_id IS NULL THEN
        RETURN NEW;
    END IF;

    SELECT EXISTS (
        SELECT 1
          FROM canonical_taxonomy_nodes
         WHERE id = NEW.parent_id
           AND namespace_id = NEW.namespace_id
           AND version_id = NEW.version_id
           AND facet_id = NEW.facet_id
    )
    INTO parent_exists;

    IF NOT parent_exists THEN
        RAISE EXCEPTION
            'canonical taxonomy parent must belong to the same namespace, version and facet';
    END IF;

    WITH RECURSIVE ancestry AS (
        SELECT
            n.id,
            n.parent_id,
            1 AS depth
        FROM canonical_taxonomy_nodes n
        WHERE n.id = NEW.parent_id
          AND n.namespace_id = NEW.namespace_id
          AND n.version_id = NEW.version_id
          AND n.facet_id = NEW.facet_id

        UNION ALL

        SELECT
            p.id,
            p.parent_id,
            a.depth + 1
        FROM canonical_taxonomy_nodes p
        JOIN ancestry a
          ON p.id = a.parent_id
        WHERE p.namespace_id = NEW.namespace_id
          AND p.version_id = NEW.version_id
          AND p.facet_id = NEW.facet_id
          AND a.depth <= 12
    )
    SELECT
        COALESCE(bool_or(id = NEW.id), false),
        COALESCE(max(depth), 0)
    INTO cycle_found, ancestor_depth
    FROM ancestry;

    IF cycle_found THEN
        RAISE EXCEPTION
            'canonical taxonomy hierarchy cycle detected';
    END IF;

    IF ancestor_depth > 12 THEN
        RAISE EXCEPTION
            'canonical taxonomy hierarchy exceeds maximum depth 12';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER canonical_taxonomy_node_guard_trg
BEFORE INSERT OR UPDATE OR DELETE
ON canonical_taxonomy_nodes
FOR EACH ROW
EXECUTE FUNCTION canonical_taxonomy_node_guard();

CREATE OR REPLACE FUNCTION canonical_taxonomy_validate_mapping()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    version_namespace TEXT;
    version_status TEXT;

    node_namespace TEXT;
    node_version TEXT;
    node_status TEXT;
BEGIN
    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;

    NEW.source_type := upper(btrim(NEW.source_type));
    NEW.source_ref := btrim(NEW.source_ref);

    IF NEW.branch_ref IS NOT NULL THEN
        NEW.branch_ref := NULLIF(btrim(NEW.branch_ref), '');
    END IF;

    IF NEW.business_ref IS NOT NULL THEN
        NEW.business_ref := NULLIF(btrim(NEW.business_ref), '');
    END IF;

    SELECT namespace_id, status
      INTO version_namespace, version_status
      FROM canonical_taxonomy_versions
     WHERE id=NEW.version_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION
            'taxonomy mapping version not found';
    END IF;

    IF version_namespace <> NEW.namespace_id THEN
        RAISE EXCEPTION
            'taxonomy mapping namespace/version ownership invalid';
    END IF;

    IF version_status <> 'PUBLISHED' THEN
        RAISE EXCEPTION
            'taxonomy mappings may target only published taxonomy versions';
    END IF;

    SELECT namespace_id, version_id, status
      INTO node_namespace, node_version, node_status
      FROM canonical_taxonomy_nodes
     WHERE id=NEW.node_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION
            'taxonomy mapping node not found';
    END IF;

    IF node_namespace <> NEW.namespace_id
       OR node_version <> NEW.version_id THEN
        RAISE EXCEPTION
            'taxonomy mapping node ownership invalid';
    END IF;

    IF node_status <> 'ACTIVE' THEN
        RAISE EXCEPTION
            'taxonomy mapping requires ACTIVE node';
    END IF;

    IF TG_OP = 'UPDATE' THEN
        IF NEW.id <> OLD.id
           OR NEW.namespace_id <> OLD.namespace_id
           OR NEW.version_id <> OLD.version_id
           OR NEW.node_id <> OLD.node_id
           OR NEW.source_type <> OLD.source_type
           OR NEW.source_ref <> OLD.source_ref
           OR NEW.mapping_type <> OLD.mapping_type THEN
            RAISE EXCEPTION
                'taxonomy mapping identity is immutable; archive and create another mapping';
        END IF;

        IF NEW.revision <> OLD.revision THEN
            RAISE EXCEPTION
                'taxonomy mapping revision is server-managed';
        END IF;

        NEW.revision := OLD.revision + 1;
    END IF;

    NEW.updated_at := now();

    RETURN NEW;
END;
$$;

CREATE TRIGGER canonical_taxonomy_mapping_validate
BEFORE INSERT OR UPDATE
ON canonical_taxonomy_mappings
FOR EACH ROW
EXECUTE FUNCTION canonical_taxonomy_validate_mapping();

// Package staffusecases implements Staff IAM business rules: provisioning,
// suspension, role/permission management, and the authorization evaluator
// every future business feature will call.
//
// The evaluator asks only "does this actor hold this permission at this
// scope" â€” never "is this actor Branch Manager." Role names exist only for
// management/usability; permissions are the sole authorization authority
// (Section 5 non-negotiable rule 16).
package staffusecases

import (
	"context"

	staffrepositories "cosmicforge/logistics/services/vendor-service/internal/features/staff/repositories"
)

// Target narrows a permission check to a branch. A nil BranchID means "no
// branch restriction" â€” appropriate for vendor-wide operations (e.g. reading
// the business's own settings). Checking a branch-scoped operation without a
// Target is a caller bug, not a security decision this evaluator should paper
// over â€” every branch-scoped route must supply one.
type Target struct {
	BranchID *string
}

// Decision is the evaluator's answer, always with a Reason even when denied â€”
// useful for audit/observability, never returned to an unauthorized caller as
// detailed justification (the HTTP layer maps any deny to a generic
// "forbidden").
type Decision struct {
	Allowed bool
	Reason  string
}

// AuthorizationEvaluator resolves a staff member's effective permission set
// (from their active role assignments, joined through role_permissions) and
// answers scoped authorization questions. Deny-by-default: a staff member
// with no matching active, non-expired assignment granting the exact
// permission at a covering scope is denied, full stop â€” nothing is inferred
// from staff status, vendor ownership, or any other signal.
type AuthorizationEvaluator struct {
	assignments *staffrepositories.AssignmentRepository
}

func NewAuthorizationEvaluator(assignments *staffrepositories.AssignmentRepository) *AuthorizationEvaluator {
	return &AuthorizationEvaluator{assignments: assignments}
}

// EffectiveGrant is one (permission, scope) pair a staff member currently
// holds via some active assignment.
type EffectiveGrant struct {
	PermissionCode string
	BranchID       *string // nil == vendor-wide
	IsHighRisk     bool
}

// EffectivePermissions returns every permission a staff member currently
// holds, each annotated with the branch scope that granted it. Callers that
// only need a yes/no answer should prefer Authorize; this is for
// "what can this actor do" listings (e.g. rendering a permissions summary).
func (e *AuthorizationEvaluator) EffectivePermissions(ctx context.Context, staffID string) ([]EffectiveGrant, error) {
	rows, err := e.assignments.EffectivePermissions(ctx, staffID)
	if err != nil {
		return nil, err
	}
	grants := make([]EffectiveGrant, 0, len(rows))
	for _, row := range rows {
		grants = append(grants, EffectiveGrant{PermissionCode: row.PermissionCode, BranchID: row.BranchID, IsHighRisk: row.IsHighRisk})
	}
	return grants, nil
}

// Authorize reports whether staffID holds permissionCode covering target.
//
// A vendor-wide grant (BranchID nil on the grant) covers every target,
// including a branch-scoped one. A branch-scoped grant only covers a target
// with the exact same BranchID â€” it never covers a vendor-wide target
// (target.BranchID nil) and never covers a different branch. This is what
// makes "a Branch Manager cannot gain another branch merely by submitting its
// branch ID" true: the grant itself is scoped at assignment time, not at
// request time.
func (e *AuthorizationEvaluator) Authorize(ctx context.Context, staffID, permissionCode string, target Target) (Decision, error) {
	if staffID == "" || permissionCode == "" {
		return Decision{Allowed: false, Reason: "missing_scope"}, nil
	}
	grants, err := e.EffectivePermissions(ctx, staffID)
	if err != nil {
		return Decision{}, err
	}
	for _, grant := range grants {
		if grant.PermissionCode != permissionCode {
			continue
		}
		if grant.BranchID == nil {
			return Decision{Allowed: true, Reason: "vendor_wide_grant"}, nil
		}
		if target.BranchID != nil && *target.BranchID == *grant.BranchID {
			return Decision{Allowed: true, Reason: "branch_scoped_grant"}, nil
		}
	}
	return Decision{Allowed: false, Reason: "no_covering_grant"}, nil
}

// HasPermission is a convenience helper over an already-fetched grant list â€”
// used where a caller has already loaded EffectivePermissions once and wants
// a plain vendor-wide-or-any-branch yes/no without a second query (e.g.
// "does this actor hold vendor.roles.manage at all, at any scope").
func HasPermission(grants []EffectiveGrant, permissionCode string) bool {
	for _, g := range grants {
		if g.PermissionCode == permissionCode {
			return true
		}
	}
	return false
}

func logConnectivity(ctx context.Context, redisAddr string, db interface{ Ping(context.Context) error }, redisClient interface {
	Ping(context.Context) *redis.StatusCmd
}) {
	pingCtx, cancel := context.WithTimeout(ctx, 5*time.Second)
	defer cancel()

	if err := db.Ping(pingCtx); err != nil {
		logging.Error("database", "connectivity check failed: %v", err)
	} else {
		logging.Success("database", "connected")
	}

	if err := redisClient.Ping(pingCtx).Err(); err != nil {
		logging.Error("redis", "failed addr=%s err=%v", redisAddr, err)
	} else {
		logging.Success("redis", "connected addr=%s", redisAddr)
	}
}

func parseDiscoveryQuery(c *gin.Context) (DiscoveryOptions, error) {

		raw := strings.TrimSpace(c.Query(name))
		if raw == "" {
			return nil, nil
		}
		value, err := strconv.ParseInt(raw, 10, 64)
		if err != nil {
			return nil, apperrors.Validation("Check your details.", []apperrors.FieldViolation{{Field: name, Message: "Value must be an integer."}})
		}
		return &value, nil
	}
	parseInt := func(name string) (*int, error) {
		raw := strings.TrimSpace(c.Query(name))
		if raw == "" {
			return nil, nil
		}
		value, err := strconv.Atoi(raw)
		if err != nil {
			return nil, apperrors.Validation("Check your details.", []apperrors.FieldViolation{{Field: name, Message: "Value must be an integer."}})
		}
		return &value, nil
	}
	minPrice, err := parseInt64("min_price_minor")
	if err != nil {
		return DiscoveryOptions{}, err
	}
	maxPrice, err := parseInt64("max_price_minor")
	if err != nil {
		return DiscoveryOptions{}, err
	}
	prepMax, err := parseInt("prep_max_minutes")
	if err != nil {
		return DiscoveryOptions{}, err
	}
	limit, err := strconv.Atoi(c.DefaultQuery("limit", "100"))
	if err != nil {
		return DiscoveryOptions{}, apperrors.Validation("Check your details.", []apperrors.FieldViolation{{Field: "limit", Message: "Value must be an integer."}})
	}
	offset, err := strconv.Atoi(c.DefaultQuery("offset", "0"))
	if err != nil {
		return DiscoveryOptions{}, apperrors.Validation("Check your details.", []apperrors.FieldViolation{{Field: "offset", Message: "Value must be an integer."}})
	}
	at := time.Time{}
	if raw := strings.TrimSpace(c.Query("at")); raw != "" {
		at, err = time.Parse(time.RFC3339, raw)
		if err != nil {
			return DiscoveryOptions{}, apperrors.Validation("Check your details.", []apperrors.FieldViolation{{Field: "at", Message: "Time must be RFC3339."}})
		}
	}
	return DiscoveryOptions{BranchID: c.Param("branchId"), Capability: c.Query("capability"), MenuID: c.Query("menu_id"), SectionID: c.Query("section_id"), Text: c.Query("q"), DietaryCode: c.Query("dietary_code"), ExcludeAllergen: c.Query("exclude_allergen"), MinPriceMinor: minPrice, MaxPriceMinor: maxPrice, PrepMaxMinutes: prepMax, SellableOnly: true, Limit: limit, Offset: offset, At: at}, nil

// routeKey is the canonical "METHOD /path" form, with gin's :param syntax
// normalised to the OpenAPI {param} form so the two surfaces are comparable.
func routeKey(method, path string) string {
	segments := strings.Split(path, "/")
	for index, segment := range segments {
		if strings.HasPrefix(segment, ":") {
			segments[index] = "{" + strings.TrimPrefix(segment, ":") + "}"
		}
	}
	return method + " " + strings.Join(segments, "/")
}

// TestEveryProductionRouteIsDocumented is the drift guard in the direction that
// matters most: a live route nobody wrote down.
func TestEveryProductionRouteIsDocumented(t *testing.T) {
	routes := productionRoutes(t)
	documented := documentedOperations(t)
	register := routeRegisterPaths(t)

	var missingSpec, missingRegister []string
	for route := range routes {
		method, path, _ := strings.Cut(route, " ")
		// Health, readiness and meta come free from serviceapp and are the
		// platform's contract rather than this service's; every other route is
		// this service's to document.
		if isPlatformRoute(path) {
			continue
		}
		if !documented[route] {
			missingSpec = append(missingSpec, route)
		}
		if !register[path] {
			missingRegister = append(missingRegister, method+" "+path)
		}
	}
	sort.Strings(missingSpec)
	sort.Strings(missingRegister)
	if len(missingSpec) != 0 {
		t.Errorf("%d live routes appear in no OpenAPI specification:\n%s",
			len(missingSpec), strings.Join(missingSpec, "\n"))
	}
	if len(missingRegister) != 0 {
		t.Errorf("%d live routes are absent from API_ROUTE_REGISTER.md:\n%s",
			len(missingRegister), strings.Join(missingRegister, "\n"))
	}
}

const specDir = "../../docs/openapi"

type specDocument struct {
	Servers []struct {
		URL string `json:"url"`
	} `json:"servers"`
	Paths map[string]map[string]json.RawMessage `json:"paths"`
}

func isMethod(value string) bool {
	switch value {
	case "get", "post", "put", "patch", "delete":
		return true
	default:
		return false
	}
}

// isPlatformRoute names the endpoints serviceapp mounts for every service.
func isPlatformRoute(path string) bool {
	switch path {
	case "/health", "/ready", "/api/v1/vendor/meta":
		return true
	}
	return false
}

// documentedOperationSources maps each documented route to the spec that
// declares it, so a phantom operation can be reported with its source.
func documentedOperationSources(t *testing.T) map[string]string {
	t.Helper()
	out := map[string]string{}
	entries, err := os.ReadDir(specDir)
	if err != nil {
		t.Fatalf("read openapi directory: %v", err)
	}
	specs := 0
	for _, entry := range entries {
		if entry.IsDir() || !strings.HasSuffix(entry.Name(), ".json") {
			continue
		}
		specs++
		raw, err := os.ReadFile(filepath.Join(specDir, entry.Name()))
		if err != nil {
			t.Fatalf("read %s: %v", entry.Name(), err)
		}
		var document specDocument
		if err = json.Unmarshal(raw, &document); err != nil {
			t.Fatalf("parse %s: %v", entry.Name(), err)
		}
		base := ""
		if len(document.Servers) > 0 {
			base = strings.TrimRight(document.Servers[0].URL, "/")
		}
		for path, methods := range document.Paths {
			for method := range methods {
				if !isMethod(method) {
					continue
				}
				full := path
				// A spec whose path already carries the internal root is
				// mounted there rather than under the server base.
				if !strings.HasPrefix(path, "/internal/") {
					full = base + path
				}
				out[strings.ToUpper(method)+" "+full] = entry.Name()
			}
		}
	}
	if specs == 0 {
		t.Fatal("no OpenAPI specifications were found")
	}
	return out
}

func documentedOperations(t *testing.T) map[string]bool {
	t.Helper()
	out := map[string]bool{}
	for route := range documentedOperationSources(t) {
		out[route] = true
	}
	return out
}

// routeRegisterPaths reads the paths named in the route register.
//
// The register is prose with tables, and it records paths in two syntaxes: the
// earlier sections use gin's `:param` form, the later ones OpenAPI's `{param}`.
// Both are normalised to one shape here so the comparison measures real drift
// rather than which decade a row was written in. The parse is otherwise
// deliberately loose â€” it asks only whether each live route is mentioned
// somewhere, because a stricter parse would fail on table formatting.
var registerPathPattern = regexp.MustCompile(`(/(?:api/v1/vendor|internal|auth|staff|roles|permissions|businesses|onboarding|assignments|crm)[A-Za-z0-9_\-/{}:\.]*)`)

func normalizeRegisterPath(path string) string {
	segments := strings.Split(path, "/")
	for index, segment := range segments {
		if strings.HasPrefix(segment, ":") {
			segments[index] = "{" + strings.TrimPrefix(segment, ":") + "}"
		}
	}
	return strings.Join(segments, "/")
}

func routeRegisterPaths(t *testing.T) map[string]bool {
	t.Helper()
	raw, err := os.ReadFile("../../docs/API_ROUTE_REGISTER.md")
	if err != nil {
		t.Fatalf("read route register: %v", err)
	}
	text := string(raw)
	out := map[string]bool{}
	record := func(path string) {
		path = normalizeRegisterPath(strings.TrimRight(path, ".,;)`"))
		if path == "" {
			return
		}
		out[path] = true
		// Most rows record a CRM or business path relative to the vendor base.
		if !strings.HasPrefix(path, "/api/") && !strings.HasPrefix(path, "/internal/") {
			out["/api/v1/vendor"+path] = true
		}
	}
	for _, match := range registerPathPattern.FindAllString(text, -1) {
		record(match)
	}
	for _, line := range strings.Split(text, "\n") {
		for _, field := range strings.Split(line, "|") {
			if field = strings.TrimSpace(field); strings.HasPrefix(field, "/") {
				record(field)
			}
		}
	}
	return out
}

func doJSON(t *testing.T, router *gin.Engine, method, path string, body any, bearer string) *httptest.ResponseRecorder {

	var reader *bytes.Reader
	if body != nil {
		b, err := json.Marshal(body)
		if err != nil {
			t.Fatalf("marshal body: %v", err)
		}
		reader = bytes.NewReader(b)
	} else {
		reader = bytes.NewReader(nil)
	}
	req := httptest.NewRequest(method, path, reader)
	req.Header.Set("Content-Type", "application/json")
	if bearer != "" {
		req.Header.Set("Authorization", "Bearer "+bearer)

	rec := httptest.NewRecorder()
	router.ServeHTTP(rec, req)
	return rec
}

	// staffhttp.RegisterInternalRoutes being reachable from cmd/main.go,
	// already proven by `go build ./...` succeeding on that wiring).
	staffRepo := staffrepositories.NewStaffRepository(pool)

	user, err := users.UpsertByEmail(context.Background(), "http-owner@example.com")

		Data struct {
			ChallengeID string `json:"challenge_id"`
			DebugOTP    string `json:"debug_otp"`
		} `json:"data"`
	}
	_ = json.Unmarshal(start.Body.Bytes(), &startEnvelope)

	verify := doJSON(t, router, http.MethodPost, "/api/v1/vendor/auth/verify", map[string]string{
		"email": "http-owner@example.com", "otp": startEnvelope.Data.DebugOTP, "challenge_id": startEnvelope.Data.ChallengeID,
	}, "")
	if verify.Code != http.StatusOK {
		t.Fatalf("expected verify 200, got %d: %s", verify.Code, verify.Body.String())
	}
	var verifyEnvelope struct {
		Data struct {
			AccessToken string `json:"access_token"`
		} `json:"data"`
	}
	_ = json.Unmarshal(verify.Body.Bytes(), &verifyEnvelope)
	if verifyEnvelope.Data.AccessToken == "" {
		t.Fatalf("expected an access token, got %s", verify.Body.String())
	}

	list := doJSON(t, router, http.MethodGet, "/api/v1/vendor/staff", nil, verifyEnvelope.Data.AccessToken)
	if list.Code != http.StatusOK {
		t.Fatalf("expected list-staff 200 with a valid token, got %d: %s", list.Code, list.Body.String())
	}
	var listEnvelope struct {
		Data struct {
			Staff []struct {
				ID string `json:"id"`
			} `json:"staff"`
		} `json:"data"`
	}
	_ = json.Unmarshal(list.Body.Bytes(), &listEnvelope)
	if len(listEnvelope.Data.Staff) != 1 || listEnvelope.Data.Staff[0].ID != "staff-http-1" {
		t.Fatalf("unexpected staff list: %s", list.Body.String())
	}

// TestHTTP_EverySection8BRouteRequiresAuthentication is the real route-parity
// guard for Section 8B: it counts what the production registration helper
// actually mounts, so the number in internal/section8bcontract cannot drift
// away from the code, and it proves not one of those routes is reachable
// without a session.
func TestHTTP_EverySection8BRouteRequiresAuthentication(t *testing.T) {
	pool := contractTestPool(t)
	redisClient := contractTestRedis(t)
	router := buildTestRouter(t, pool, redisClient)

	checked := 0
	for _, route := range router.Routes() {
		if !strings.HasPrefix(route.Path, "/api/v1/vendor/crm/tags") &&
			!strings.HasPrefix(route.Path, "/api/v1/vendor/crm/segments") &&
			route.Path != "/api/v1/vendor/crm/segment-fields" {
			continue
		}
		path := route.Path
		for _, name := range []string{":tagId", ":segmentId"} {
			path = strings.ReplaceAll(path, name, "unknown")
		}
		rec := doJSON(t, router, route.Method, path, map[string]any{}, "")
		if rec.Code != http.StatusUnauthorized {
			t.Errorf("%s %s without auth=%d body=%s", route.Method, path, rec.Code, rec.Body.String())
		}
		checked++
	}
	if checked != 21 {
		t.Fatalf("checked %d Section 8B routes, want 21", checked)
	}
}
